1| /**
2|  * Heatmap texture generation — IDW scalar field interpolation
3|  *
4|  * APPROACH (same as Windy, Copernicus, NOAA):
5|  *   1. Build a coarse value grid using Inverse Distance Weighting (IDW)
6|  *      interpolation from the sparse data points.
7|  *   2. Map each interpolated value through an absolute colour scale.
8|  *   3. Write RGBA pixels directly into an ImageData buffer — no canvas
9|  *      compositing, no alpha blending between blobs.  Each pixel gets exactly
10|  *      the colour matching its interpolated value.
11|  *   4. Apply a single Gaussian blur pass on the 3-wide tiled canvas to
12|  *      smooth the field without introducing seams at lon = ±180.
13|  *
14|  * This means:
15|  *   - 11 m/s wind → teal/green, NOT purple
16|  *   - 37°C        → red/orange
17|  *   - AQI 159     → red
18|  *   - No colour mixing from overlapping blobs (there are no blobs)
19|  *
20|  * COORDINATE SYSTEM (equirectangular, Three.js SphereGeometry UV):
21|  *   pixel x = (lon + 180) / 360 * W
22|  *   pixel y = (90  - lat) / 180 * H
23|  */
24| 
25| import type {
26|   WindPoint,
27|   TempAnomalyPoint,
28|   AQIPoint,

35| const W = 2048; // texture width  (equirectangular)
36| const H = 1024; // texture height

38| /**
39|  * IDW grid resolution — we compute the interpolated value on a downsampled
40|  * grid, then upsample with bilinear interpolation for smooth output.
41|  *
42|  * IDW_STEP = 8  →  256×128 = 32,768 cells
43|  * With ~2800 wind points: 32,768 × 2800 = ~92M ops — fast enough for a
44|  * one-time texture build on modern hardware (~200ms).
45|  * K = 4 nearest neighbours is sufficient for smooth IDW on a regular grid.
46|  */

48| const IDW_GW = Math.ceil(W / IDW_STEP) + 1; // +1 so bilinear can always read [gx+1]
49| const IDW_GH = Math.ceil(H / IDW_STEP) + 1;

53| function lerp3(
54|   a: [number, number, number],
55|   b: [number, number, number],
56|   t: number,
57| ): [number, number, number] {
58|   return [
59|     Math.round(a[0] + (b[0] - a[0]) * t),
60|     Math.round(a[1] + (b[1] - a[1]) * t),
61|     Math.round(a[2] + (b[2] - a[2]) * t),
62|   ];
63| }

65| function gradientColor(
66|   t: number,
67|   stops: [number, number, number][],

69|   const clamped = Math.max(0, Math.min(1, t));
70|   const seg = clamped * (stops.length - 1);
71|   const idx = Math.min(Math.floor(seg), stops.length - 2);
72|   return lerp3(stops[idx], stops[idx + 1], seg - idx);

82| // ─── IDW interpolation ────────────────────────────────────────────────────────

84| /**
85|  * Build a W/IDW_STEP × H/IDW_STEP Float32Array of interpolated values using
86|  * Inverse Distance Weighting.
87|  *
88|  * For each grid cell we find the K nearest data points (by squared pixel
89|  * distance) and weight them as 1/d².  Using pixel distance means the
90|  * influence of each point scales correctly with the projection.
91|  *
92|  * power = 2 gives smooth gradients; increase to sharpen local features.
93|  */
94| function buildIDWGrid(
95|   points: { px: number; py: number; value: number }[],
96|   K = 4,
97| ): Float32Array {
98|   const grid = new Float32Array(IDW_GW * IDW_GH);
99|   const KMAX = Math.min(K, points.length);
100| 
101|   for (let gy = 0; gy < IDW_GH; gy++) {
102|     for (let gx = 0; gx < IDW_GW; gx++) {
103|       const cx = gx * IDW_STEP;
104|       const cy = gy * IDW_STEP;
105| 
106|       const bestD = new Float32Array(KMAX).fill(Infinity);
107|       const bestV = new Float32Array(KMAX);
108|       let exactVal = NaN;
109| 
110|       for (let i = 0; i < points.length; i++) {
111|         const dx = cx - points[i].px;
112|         const dy = cy - points[i].py;
113|         const d2 = dx * dx + dy * dy;
114| 
115|         if (d2 < 1) {
116|           exactVal = points[i].value;
117|           break;
118|         } // exact hit
119| 
120|         if (d2 < bestD[KMAX - 1]) {
121|           bestD[KMAX - 1] = d2;
122|           bestV[KMAX - 1] = points[i].value;
123|           // Insertion sort (ascending) to keep smallest distances first
124|           for (let j = KMAX - 1; j > 0 && bestD[j] < bestD[j - 1]; j--) {
125|             let tmp = bestD[j];
126|             bestD[j] = bestD[j - 1];
127|             bestD[j - 1] = tmp;
128|             tmp = bestV[j];
129|             bestV[j] = bestV[j - 1];
130|             bestV[j - 1] = tmp;
131|           }
132|         }
133|       }
134| 
135|       if (!isNaN(exactVal)) {
136|         grid[gy * IDW_GW + gx] = exactVal;
137|         continue;
138|       }
139| 
140|       let sumW = 0,
141|         sumWV = 0;
142|       for (let k = 0; k < KMAX; k++) {
143|         if (bestD[k] === Infinity) break;
144|         const w = 1 / bestD[k]; // power=2, but d2 already squared → w = 1/d²
145|         sumW += w;
146|         sumWV += w * bestV[k];
147|       }
148|       grid[gy * IDW_GW + gx] = sumW > 0 ? sumWV / sumW : 0;
149|     }
150|   }
151| 
152|   return grid;
153| }

157| /**
158|  * Convert an IDW value grid into full-resolution RGBA using bilinear
159|  * interpolation between grid cells — avoids the blocky nearest-neighbour
160|  * artefacts that would otherwise show at IDW_STEP=8.
161|  */

163|   grid: Float32Array,
164|   colorFn: (v: number) => [number, number, number],
165| ): ImageData {
166|   const imgData = new ImageData(W, H);
167|   const buf = imgData.data;
168| 
169|   for (let py = 0; py < H; py++) {

171|     const gy0 = Math.min(Math.floor(fy), IDW_GH - 2);
172|     const gy1 = gy0 + 1;

176|       const fx = px / IDW_STEP;

178|       const gx1 = gx0 + 1;

207|       const d = (p.lat - lat) ** 2 + (p.lon - lon) ** 2;

209|     }
210|     return {
211|       type: "aqi", lat, lon, aqi: Math.round(val),
212|       pm25: nearest?.pm25 ?? 0, category: nearest?.category ?? "Unknown",
213|     };
214|   }
215|   if (layer === "sea_temp") {
216|     return { type: "sea_temp", lat, lon, tempC: val };
217|   }
218|   return null;
219| }
