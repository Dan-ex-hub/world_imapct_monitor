444|               className="rounded-lg p-1.5 text-text-muted transition-colors hover:bg-bg-elevated hover:text-text-primary"
445|               title="Exit playback"
446|             >
447|               <X className="h-4 w-4" />
448|             </button>
449|           </div>
450| 
451|           {/* ── Timeline ──────────────────────────────────────────────────── */}
452|           {/* Event markers above the scrubber */}
453|           <div className="relative mb-1 h-3">
454|             {allEventsSnap.map((e) => {
455|               const range = endTime.getTime() - startTime.getTime();
456|               if (range <= 0) return null;
457|               const pct =
458|                 ((new Date(e.publishedAt).getTime() - startTime.getTime()) /
459|                   range) *
460|                 100;
461|               if (pct < 0 || pct > 100) return null;
462|               const revealed =
463|                 new Date(e.publishedAt).getTime() <= clock.getTime();
464|               return (
465|                 <div
466|                   key={e.id}
467|                   title={`${e.headline} — ${format(new Date(e.publishedAt), "HH:mm")}`}

469|                   style={{
470|                     left: `${pct}%`,
471|                     backgroundColor: revealed
472|                       ? e.impactLevel === "Critical"
473|                         ? "#ef4444"
474|                         : e.impactLevel === "High"
475|                           ? "#f97316"
476|                           : e.impactLevel === "Medium"
477|                             ? "#eab308"
478|                             : "#22c55e"
479|                       : "rgba(255,255,255,0.15)",
480|                     transform: `translateX(-50%) scale(${revealed ? 1 : 0.7})`,
481|                   }}
482|                   onClick={() => seekTo(new Date(e.publishedAt))}
483|                 />
484|               );
485|             })}
486|           </div>
487| 
488|           {/* Scrubber bar */}
489|           <div
490|             className="mb-3 h-1.5 w-full cursor-pointer rounded-full bg-bg-elevated"
491|             onClick={handleScrub}
492|             title="Drag to seek"
493|           >
494|             <div
495|               className="h-full rounded-full bg-impact-medium transition-none"
496|               style={{ width: `${progress}%` }}
497|             />
498|           </div>
499| 
500|           {/* ── Transport row ──────────────────────────────────────────────── */}
501|           <div className="flex items-center justify-between">
502|             {/* Start label */}
503|             <span className="w-24 text-[10px] text-text-muted">
504|               {format(startTime, "MMM d HH:mm")}
505|             </span>
506| 
507|             {/* Controls */}
508|             <div className="flex items-center gap-1">
509|               <button
510|                 onClick={restart}
511|                 className="rounded-lg p-2 text-text-muted transition-colors hover:bg-bg-elevated hover:text-text-primary"
512|                 title="Restart from beginning"
513|               >
514|                 <RotateCcw className="h-3.5 w-3.5" />
515|               </button>
516| 
517|               <button
518|                 onClick={skipBack}
519|                 disabled={clock <= startTime}
520|                 className="rounded-lg p-2 text-text-muted transition-colors hover:bg-bg-elevated hover:text-text-primary disabled:opacity-30"
521|                 title="Back 1 hour"
522|               >
523|                 <SkipBack className="h-3.5 w-3.5" />
524|               </button>
525| 
526|               <button
527|                 onClick={() => setIsPlaying((p) => !p)}
528|                 disabled={isAtEnd}
529|                 className="mx-1 rounded-xl bg-impact-medium px-5 py-2 text-white transition-colors hover:bg-impact-medium/80 disabled:opacity-50"
530|                 title={isPlaying ? "Pause" : "Play"}
531|               >
532|                 {isPlaying ? (
533|                   <Pause className="h-4 w-4" />
534|                 ) : (
535|                   <Play className="h-4 w-4" />
536|                 )}
537|               </button>
538| 
539|               <button
540|                 onClick={skipForward}
541|                 disabled={isAtEnd}
542|                 className="rounded-lg p-2 text-text-muted transition-colors hover:bg-bg-elevated hover:text-text-primary disabled:opacity-30"
543|                 title="Forward 1 hour"
544|               >
545|                 <SkipForward className="h-3.5 w-3.5" />
546|               </button>
547|             </div>
548| 
549|             {/* End label */}
550|             <span className="w-24 text-right text-[10px] text-text-muted">
551|               {format(endTime, "MMM d HH:mm")}
552|             </span>
553|           </div>
554| 
555|           {/* ── Caught-up message ──────────────────────────────────────────── */}
556|           {isAtEnd && (
557|             <div className="mt-3 flex items-center justify-center gap-2 text-[11px] text-text-muted">
558|               <span>All {totalCount} events replayed —</span>
559|               <button
560|                 onClick={restart}
561|                 className="font-medium text-impact-medium underline underline-offset-2"
562|               >
563|                 replay again
564|               </button>
565|             </div>
566|           )}
567|         </div>
568|       </div>
569|     </>
570|   );
