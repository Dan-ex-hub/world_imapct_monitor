19| // ─── EPA PM2.5 → AQI conversion ───────────────────────────────────────────────
20| 
21| export function aqiFromPm25(pm25: number): number {
22|   const bp: [number, number, number, number][] = [
23|     [0, 12, 0, 50],
24|     [12.1, 35.4, 51, 100],
25|     [35.5, 55.4, 101, 150],
26|     [55.5, 150.4, 151, 200],
27|     [150.5, 250.4, 201, 300],
28|     [250.5, 350.4, 301, 400],
29|     [350.5, 500.4, 401, 500],
30|   ];
31|   for (const [cLo, cHi, iLo, iHi] of bp) {
32|     if (pm25 >= cLo && pm25 <= cHi) {
33|       return Math.round(((iHi - iLo) / (cHi - cLo)) * (pm25 - cLo) + iLo);
34|     }
35|   }
36|   return pm25 > 500 ? 500 : 0;
37| }

8| export function useRealtimeEvents() {

10|   const addEvent = useGlobeStore((s) => s.addEvent);
11| 
12|   useEffect(() => {
13|     const supabase = createClient();
14| 
15|     // Fetch events from the last 48h — this gives the globe a healthy spread

17|     const fetchInitial = async () => {
18|       const fortyEightHoursAgo = new Date(Date.now() - 48 * 60 * 60 * 1000).toISOString();
19|       const { data, error } = await supabase
20|         .from("events")
21|         .select("*")
22|         .gte("published_at", fortyEightHoursAgo)
23|         .order("published_at", { ascending: false })
24|         .limit(500);
25| 
26|       if (error) {
27|         console.error("[Realtime] fetch error:", error);
28|         return;
29|       }
30|       if (data) {
31|         setEvents(data.map(mapRow));
32|       }
33|     };
34| 
35|     fetchInitial();
36| 
37|     // Subscribe to INSERT / UPDATE — new events arrive live
38|     // Subscribe to DELETE — when /api/news/refresh clears stale rows,
39|     //   they're removed from the store immediately (no "17h ago" ghost events)
40|     const channel = supabase
41|       .channel("events-channel")
42|       .on(
43|         "postgres_changes",
44|         { event: "INSERT", schema: "public", table: "events" },
45|         (p) => addEvent(mapRow(p.new as any)),
46|       )
47|       .on(
48|         "postgres_changes",
49|         { event: "UPDATE", schema: "public", table: "events" },
50|         (p) => addEvent(mapRow(p.new as any)),
51|       )
52|       .on(
53|         "postgres_changes",
54|         { event: "DELETE", schema: "public", table: "events" },
