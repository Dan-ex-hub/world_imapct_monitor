1| /**
2|  * Simple in-memory rate limiter for API routes
3|  * Prevents hammering free external APIs
4|  */
5| 
6| interface RateLimitEntry {
7|   count: number
8|   resetAt: number
9| }
10| 
11| const rateLimitStore = new Map<string, RateLimitEntry>()
12| 
13| // Cleanup old entries every 5 minutes
14| setInterval(() => {
15|   const now = Date.now()
16|   for (const [key, entry] of rateLimitStore.entries()) {
17|     if (entry.resetAt < now) {
18|       rateLimitStore.delete(key)
19|     }
20|   }
21| }, 300_000)
22| 
23| export interface RateLimitConfig {
24|   maxRequests: number
25|   windowMs: number
26| }
27| 
28| /**
29|  * Check if a request should be rate limited
30|  * @param identifier - Unique identifier (e.g., IP address or route name)
31|  * @param config - Rate limit configuration
32|  * @returns true if rate limit exceeded, false otherwise
33|  */
34| export function isRateLimited(
35|   identifier: string,
36|   config: RateLimitConfig
37| ): boolean {
38|   const now = Date.now()
39|   const entry = rateLimitStore.get(identifier)
40| 
41|   if (!entry || entry.resetAt < now) {
42|     // First request or window expired - reset
43|     rateLimitStore.set(identifier, {
44|       count: 1,
45|       resetAt: now + config.windowMs,
46|     })
47|     return false
48|   }
49| 
50|   if (entry.count >= config.maxRequests) {
51|     // Rate limit exceeded
52|     return true
53|   }
54| 
55|   // Increment count
56|   entry.count++
57|   return false
58| }
59| 
60| /**
61|  * Get rate limit info for a request
62|  */
63| export function getRateLimitInfo(identifier: string) {
64|   const entry = rateLimitStore.get(identifier)
65|   if (!entry) {
66|     return { remaining: Infinity, resetAt: null }
67|   }
68| 
69|   const now = Date.now()
70|   if (entry.resetAt < now) {
71|     return { remaining: Infinity, resetAt: null }
72|   }
73| 
74|   return {
75|     remaining: Math.max(0, entry.resetAt - now),
76|     resetAt: new Date(entry.resetAt).toISOString(),
77|   }
78| }
